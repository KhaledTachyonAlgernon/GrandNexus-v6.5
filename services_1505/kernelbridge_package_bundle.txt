# Child bootstrap
def _child_bootstrap(cfg: Dict[str,Any], in_q: mp.Queue, out_q: mp.Queue):
    import psutil
    # impose RAM limit
    if cfg.get("ram_limit_mb"):
        lim = cfg["ram_limit_mb"] * 1024 * 1024
        resource.setrlimit(resource.RLIMIT_AS, (lim, lim))
    # simple echo loop
    while True:
        try:
            msg = in_q.get(timeout=1)
            if msg is None: break
            out_q.put({"child": os.getpid(), "echo": msg})
        except queue.Empty:
            pass

class SpawnPolicy(BaseModel):
    ram_limit_mb: Optional[int] = None

class KernelBridge:
    def __init__(self):
        self.logger = logging.getLogger("kernelbridge")
        self._children = {}  # pid → {proc,in_q,out_q,policy,start}
        self._lock = threading.RLock()
        self.app = FastAPI(title="KernelBridge API")
        self._configure_api()
        threading.Thread(target=self._pump_loop, daemon=True).start()

    def spawn(self, policy: SpawnPolicy) -> int:
        in_q, out_q = mp.Queue(), mp.Queue()
        cfg = {"ram_limit_mb": policy.ram_limit_mb}
        proc = mp.Process(target=_child_bootstrap, args=(cfg,in_q,out_q), daemon=True)
        proc.start()
        cid = proc.pid
        with self._lock:
            self._children[cid] = {"proc":proc,"in_q":in_q,"out_q":out_q,"start":time.time()}
        SPAWN_COUNT.inc()
        return cid

    def terminate(self, cid: int):
        with self._lock:
            info = self._children.pop(cid, None)
        if info:
            info["in_q"].put(None)
            info["proc"].join(timeout=3)
            TERMINATE_COUNT.inc()

    def _pump_loop(self):
        while True:
            with self._lock:
                for cid,info in list(self._children.items()):
                    # update uptime metric
                    CHILD_UPTIME.labels(child_id=str(cid)).set(time.time()-info["start"])
            time.sleep(cfg.STATUS_INTERVAL)

    def _configure_api(self):
        @self.app.post("/spawn")
        def api_spawn(policy: SpawnPolicy):
            cid = self.spawn(policy)
            return {"child_id": cid}

        @self.app.post("/terminate/{cid}")
        def api_terminate(cid: int):
            self.terminate(cid)
            return {"terminated": cid}

        @self.app.get("/status")
        def api_status():
            with self._lock:
                return {"children": list(self._children.keys())}
