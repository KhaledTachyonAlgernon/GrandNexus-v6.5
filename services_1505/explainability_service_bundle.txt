from pydantic import BaseSettings, Field, HttpUrl
from typing import Optional

class ExplainabilityConfig(BaseSettings):
    # LLM backend
    backend: str = Field("openai", env="EXPL_BACKEND")
    openai_api_key: Optional[str] = Field(None, env="OPENAI_API_KEY")
    openai_model: str = Field("gpt-4-turbo", env="OPENAI_MODEL")
    timeout_s: int = Field(5, env="EXPL_TIMEOUT_S")

    # Circuit breaker
    cb_max_failures: int = Field(3, env="EXPL_CB_MAX_FAILURES")
    cb_reset_timeout_s: int = Field(60, env="EXPL_CB_RESET_TIMEOUT_S")

    # API
    api_host: str = Field("0.0.0.0", env="EXPL_HOST")
    api_port: int = Field(8081, env="EXPL_PORT")

    class Config:
        env_file = ".env"

from prometheus_client import Counter, Gauge

REQ_COUNT = Counter("expl_requests_total", "Total explanation requests")
REQ_FAILURE = Counter("expl_failures_total", "Failed explanation requests")
LATENCY = Gauge("expl_request_latency_seconds", "Latency per explanation request")

import time, threading
from typing import Dict, Any
from tenacity import retry, stop_after_attempt, wait_fixed, retry_if_exception_type, RetryError
from circuitbreaker import circuit, CircuitBreakerError
import openai
from .config import ExplainabilityConfig
from .metrics import REQ_COUNT, REQ_FAILURE, LATENCY

cfg = ExplainabilityConfig()

# Configure OpenAI
if cfg.backend == "openai":
    openai.api_key = cfg.openai_api_key

# Circuit breaker decorator
@circuit(failure_threshold=cfg.cb_max_failures,
         recovery_timeout=cfg.cb_reset_timeout_s)
def _call_llm(prompt: str) -> str:
    resp = openai.ChatCompletion.create(
        model=cfg.openai_model,
        messages=[{"role":"user","content":prompt}],
        timeout=cfg.timeout_s
    )
    return resp.choices[0].message.content

@retry(stop=stop_after_attempt(2), wait=wait_fixed(1), retry=retry_if_exception_type(Exception))
def generate_explanation(context: Dict[str,Any], decision: Dict[str,Any]) -> str:
    prompt = (
        f"Context:\n{context}\n\n"
        f"Decision:\n{decision}\n\n"
        "Explain why this decision was made."
    )
    start = time.time()
    try:
        REQ_COUNT.inc()
        text = _call_llm(prompt)
        return text
    except (CircuitBreakerError, RetryError) as e:
        REQ_FAILURE.inc()
        return "Explication indisponible pour le moment."
    finally:
        LATENCY.set(time.time() - start)

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import uvicorn
from .service import generate_explanation
from .config import cfg

app = FastAPI(title="ExplainabilityEngine API")

class ExplRequest(BaseModel):
    context: dict
    decision: dict

class ExplResponse(BaseModel):
    explanation: str

@app.post("/explain", response_model=ExplResponse)
def explain(req: ExplRequest):
    try:
        text = generate_explanation(req.context, req.decision)
        return ExplResponse(explanation=text)
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

def start_api():
    uvicorn.run(app, host=cfg.api_host, port=cfg.api_port)
